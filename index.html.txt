            <div id="live-datetime" class="text-gray-500 text-xs font-mono mt-1"></div>
        </div>
        <div class="flex gap-2 mt-3 sm:mt-0 flex-wrap justify-center">
            <button onclick="authAction('addBox')" class="bg-green-600 text-white px-3 py-2 rounded font-semibold text-xs shadow">➕ Add Box</button>
            <button onclick="authAction('removeBox')" class="bg-red-600 text-white px-3 py-2 rounded font-semibold text-xs shadow">➖ Remove Box</button>

                <button onclick="closeModal('modal-box-menu')" class="text-red-500 font-bold text-xl">&times;</button>
            </div>
            
            <input type="hidden" id="edit-box-id">
            
            <div class="bg-gray-50 p-3 rounded border mb-3">
                <h3 class="font-bold text-sm mb-2 text-gray-600">Current/New Game Details</h3>
                <div class="flex gap-2 mb-2">
                    <input type="text" id="game-id" placeholder="Game ID" class="w-1/2 border p-2 rounded text-sm">
                    <button onclick="openScanner('game-id')" class="bg-gray-200 px-3 rounded">📷</button>
                </div>
                <div class="flex gap-2 mb-2">
                    <input type="number" id="game-price" placeholder="Price $" class="w-full border p-2 rounded text-sm">
                </div>
                <div class="flex gap-2 mb-2">
                    <input type="number" id="start-ticket" placeholder="Start Ticket" class="w-1/2 border p-2 rounded text-sm">
                    <input type="number" id="end-ticket" placeholder="End Ticket" class="w-1/2 border p-2 rounded text-sm">
                </div>
            </div>

            <div class="flex flex-col gap-2">
                <button onclick="saveRollData(false)" class="bg-blue-600 text-white py-2 rounded font-bold">💾 Save & Update Active Game</button>
                <button onclick="saveRollData(true)" class="bg-purple-600 text-white py-2 rounded font-bold border-t border-purple-800 mt-2">🔄 Swap: Mark OLD & Load New Game</button>
            </div>
        </div>

    <div id="modal-scanner" class="modal">
        <div class="bg-white p-3 rounded-lg w-11/12 max-w-md flex flex-col items-center">
            <h2 class="text-lg font-bold mb-2">Scan Barcode</h2>
            <div class="bg-gray-200 w-full h-48 flex items-center justify-center mb-3 rounded border-2 border-dashed border-gray-400">
                <span class="text-gray-500">[Camera Feed Placeholder]</span>
            </div>
            <button onclick="simulateScan()" class="px-4 py-2 bg-green-600 text-white rounded font-bold w-full mb-2">Simulate Successful Scan</button>
            <button onclick="closeModal('modal-scanner')" class="px-4 py-2 bg-red-600 text-white rounded font-bold w-full">Cancel</button>
        </div>

    <script>
        // State Management
        let appState = {
            boxes: [],
            totalScanned: 0
        };

        let currentScanTarget = '';

        function initApp() {
            // Live clock
            setInterval(() => {
                document.getElementById('live-datetime').innerText = new Date().toLocaleString();
            }, 1000);

            // Generate default 40 boxes
            for (let i = 1; i <= 40; i++) {
                appState.boxes.push({
                    id: i,
                    label: i.toString(),
                    rolls: [] // Array to hold games (allows multiple for mid-shift swaps)
                });
            }
            renderGrid();
        }

        function renderGrid() {
            const grid = document.getElementById('lotto-grid');
            grid.innerHTML = '';
            
            appState.boxes.forEach(box => {
                let contentHTML = `<div class="text-xs text-gray-500 text-center py-2">Empty</div>`;
                
                if (box.rolls.length > 0) {
                    contentHTML = '';
                    box.rolls.forEach((roll, index) => {
                        const isOld = box.rolls.length > 1 && index < box.rolls.length - 1;
                        const statusClass = isOld ? 'text-gray-400' : 'text-green-700 font-bold';
                        const prefix = isOld ? '[OLD]' : '[NEW]';
                        const wrapperClass = index > 0 ? 'merged-slot' : '';
                        
                        const sold = roll.end !== null ? (roll.end - roll.start) : 0;
                        const soldText = roll.end !== null ? `Sold: ${sold}` : `Start: ${roll.start}`;

                        contentHTML += `
                            <div class="${wrapperClass} text-xs">
                                <div class="flex justify-between ${statusClass}">
                                    <span>${prefix} #${roll.gameId} ($${roll.price})</span>
                                    <span>${soldText}</span>
                                </div>
                            </div>
                        `;
                    });
                }

                grid.innerHTML += `
                    <div class="box-card cursor-pointer hover:shadow-md transition" onclick="openBoxMenu(${box.id})">
                        <div class="flex justify-between items-center border-b pb-1 mb-1">
                            <span class="font-bold text-gray-700">Box ${box.label}</span>
                        </div>
                        ${contentHTML}
                    </div>
                `;
            });
            calculateMath();
        }

        function openBoxMenu(boxId) {
            const box = appState.boxes.find(b => b.id === boxId);
            document.getElementById('edit-box-id').value = boxId;
            document.getElementById('modal-box-title').innerText = `Manage Box ${box.label}`;
            
            // Load current active game data if it exists
            const activeRoll = box.rolls.length > 0 ? box.rolls[box.rolls.length - 1] : null;
            
            document.getElementById('game-id').value = activeRoll ? activeRoll.gameId : '';
            document.getElementById('game-price').value = activeRoll ? activeRoll.price : '';
            document.getElementById('start-ticket').value = activeRoll ? activeRoll.start : '';
            document.getElementById('end-ticket').value = activeRoll && activeRoll.end !== null ? activeRoll.end : '';

            document.getElementById('modal-box-menu').classList.add('active');
        }

        function saveRollData(isSwap) {
            const boxId = parseInt(document.getElementById('edit-box-id').value);
            const box = appState.boxes.find(b => b.id === boxId);
            
            const gId = document.getElementById('game-id').value || '000';
            const gPrice = parseFloat(document.getElementById('game-price').value) || 0;
            const sTick = parseInt(document.getElementById('start-ticket').value) || 0;
            const eTickRaw = document.getElementById('end-ticket').value;
            const eTick = eTickRaw !== '' ? parseInt(eTickRaw) : null;

            const newRollData = { gameId: gId, price: gPrice, start: sTick, end: eTick };

            if (isSwap) {
                // Mid-shift swap: The existing active roll gets locked, and this new data is pushed as a new game
                box.rolls.push(newRollData);
            } else {
                // Update active game
                if (box.rolls.length === 0) {
                    box.rolls.push(newRollData);
                } else {
                    box.rolls[box.rolls.length - 1] = newRollData;
                }
            }

            closeModal('modal-box-menu');
            renderGrid();
        }

        function calculateMath() {
            let total = 0;
            appState.boxes.forEach(box => {
                box.rolls.forEach(roll => {
                    if (roll.end !== null) {
                        total += (roll.end - roll.start) * roll.price;
                    }
                });
            });
            
            document.getElementById('total-scanned').innerText = `$${total.toFixed(2)}`;
            document.getElementById('shift-variance').innerText = `$${total.toFixed(2)}`; // Assuming $0 POS for test
        }

        // --- Auth & Admin Actions ---
        function authAction(action) {
            const pwd = prompt("Enter Admin Password (type 'admin'):");
            if (pwd === 'admin') {
                if (action === 'addBox') {
                    const nextId = appState.boxes.length + 1;
                    appState.boxes.push({ id: nextId, label: nextId.toString(), rolls: [] });
                } else if (action === 'removeBox') {
                    if(appState.boxes.length > 0) appState.boxes.pop();
                }
                renderGrid();
            } else {
                alert("Incorrect Password");
            }
        }

        // --- Scanner Logic ---
        function openScanner(targetInputId) {
            currentScanTarget = targetInputId;
            document.getElementById('modal-scanner').classList.add('active');
        }

        function simulateScan() {
            // Mocks a successful barcode scan inserting a Game ID
            document.getElementById(currentScanTarget).value = Math.floor(1000 + Math.random() * 9000);
            closeModal('modal-scanner');
        }

        function closeModal(id) { 
            document.getElementById(id).classList.remove('active'); 
        }
    </script>
